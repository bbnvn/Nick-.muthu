<script>

let ws = new WebSocket("ws://localhost:3000");

let state = "RUNNING";
let currentMultiplier = 1;

let countdown = 5;
let countdownInterval = null;

let x = 120, y = 200;
let vx = 4, vy = 3;

let feed = document.getElementById("feed");
let history = document.getElementById("history");
let countdownBox = document.getElementById("countdown");

function addFeed(t){
    let d = document.createElement("div");
    d.innerText = t;
    feed.prepend(d);
}

function start1(){ addFeed("Panel 1 STARTED"); }
function start2(){ addFeed("Panel 2 STARTED"); }

function cashIn1(){
    let t = parseFloat(document.getElementById("t1").value);
    addFeed(currentMultiplier>=t ?
        "P1 SUCCESS " + currentMultiplier.toFixed(2) + "x" :
        "P1 FAIL " + currentMultiplier.toFixed(2) + "x");
}

function cashIn2(){
    let t = parseFloat(document.getElementById("t2").value);
    addFeed(currentMultiplier>=t ?
        "P2 SUCCESS " + currentMultiplier.toFixed(2) + "x" :
        "P2 FAIL " + currentMultiplier.toFixed(2) + "x");
}

/* SOCKET */
ws.onmessage = (msg)=>{
    let d = JSON.parse(msg.data);

    if(d.type === "multiplier" && state === "RUNNING"){
        currentMultiplier = parseFloat(d.value);
        document.getElementById("mult").innerText = d.value + "x";
    }

    if(d.type === "crash"){

        state = "CRASHED";

        let b = document.createElement("span");
        b.className = "badge";
        b.innerText = d.value + "x";
        history.prepend(b);

        addFeed("CRASHED at " + d.value + "x");

        document.getElementById("mult").innerText = "CRASHED";

        clearInterval(countdownInterval);

        countdown = 5;
        countdownBox.innerText = "Next round in: " + countdown;

        state = "COUNTDOWN";

        countdownInterval = setInterval(()=>{

            countdown--;

            if(countdown > 0){
                countdownBox.innerText = "Next round in: " + countdown;
            }

            if(countdown === 0){

                clearInterval(countdownInterval);

                state = "RUNNING";
                currentMultiplier = 1;

                document.getElementById("mult").innerText = "1.00x";
                countdownBox.innerText = "";

                x = 120;
                y = 200;
                vx = 4;
                vy = 3;

                addFeed("NEW ROUND STARTED");
            }

        },1000);
    }
};

/* BALL */
function draw(){
    let c = document.getElementById("c");
    let ctx = c.getContext("2d");

    ctx.clearRect(0,0,c.width,c.height);

    ctx.fillStyle="red";
    ctx.beginPath();
    ctx.arc(x,y,12,0,Math.PI*2);
    ctx.fill();

    if(state === "RUNNING"){
        x += vx;
        y += vy;
    }

    if(x<20||x>880) vx*=-1;
    if(y<20||y>430) vy*=-1;

    requestAnimationFrame(draw);
}

draw();
