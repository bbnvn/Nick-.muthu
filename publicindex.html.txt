<script>
let ws = new WebSocket("ws://localhost:3000");

const canvas = document.getElementById("c");
const ctx = canvas.getContext("2d");

let x = 150;
let y = 200;

let vx = 4;
let vy = 3;

let historyDiv = document.getElementById("history");
let feedDiv = document.getElementById("feed");

let roundHistory = [];

function addFeed(text){
    let div = document.createElement("div");
    div.innerText = text;
    feedDiv.prepend(div);
}

function updateHistory(value){

    roundHistory.unshift(value);

    if(roundHistory.length > 10){
        roundHistory.pop();
    }

    historyDiv.innerHTML = "";

    roundHistory.forEach(v=>{
        let span = document.createElement("span");
        span.className = "badge";
        span.innerText = v + "x";
        historyDiv.appendChild(span);
    });
}

ws.onmessage = (msg)=>{

    let data = JSON.parse(msg.data);

    if(data.type === "multiplier"){

        document.getElementById("mult").innerText =
            data.value + "x";

    }

    if(data.type === "crash"){

        updateHistory(data.value);

        addFeed(
            "Round crashed at " +
            data.value + "x"
        );
    }

    if(data.type === "cashout_success"){

        addFeed(
            "Cashout at " +
            Number(data.multiplier).toFixed(2) +
            "x"
        );
    }
};

function playPanel1(){

    let amount =
        Number(
            document.getElementById("bet1").value
        );

    addFeed(
        "Panel 1 played with " +
        amount +
        " credits"
    );
}

function playPanel2(){

    let amount =
        Number(
            document.getElementById("bet2").value
        );

    addFeed(
        "Panel 2 played with " +
        amount +
        " credits"
    );
}

function drawBall(){

    ctx.clearRect(
        0,
        0,
        canvas.width,
        canvas.height
    );

    ctx.fillStyle = "red";

    ctx.beginPath();

    ctx.arc(
        x,
        y,
        12,
        0,
        Math.PI * 2
    );

    ctx.fill();

    x += vx;
    y += vy;

    if(x < 20 || x > canvas.width - 20){
        vx *= -1;
    }

    if(y < 20 || y > canvas.height - 20){
        vy *= -1;
    }

    requestAnimationFrame(drawBall);
}

drawBall();
